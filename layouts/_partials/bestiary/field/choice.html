{{- /* Renders the declared choice whose value matches; fails the build when none does. */ -}} {{- $value := .value -}} {{- $choice := "" -}} {{- range .field.choices -}} {{- if eq .value $value }}{{ $choice = . }}{{ end -}} {{- end -}} {{- with $choice -}} {{ .label }} {{- else -}} {{- errorf "%s: %s: %q is not one of the declared choices" .page.File.Path .field.key .value -}} {{- end -}}
