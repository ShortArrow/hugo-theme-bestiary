{{- /* Renders the declared choice whose value matches, tilted and nudged by an amount fixed per page, linked to its page when the field has pages = true; fails the build when none does. */ -}} {{- $value := .value -}} {{- $choice := "" -}} {{- range .field.choices -}} {{- if eq .value $value }}{{ $choice = . }}{{ end -}} {{- end -}} {{- $href := "" -}} {{- if and $choice .field.pages -}} {{- with site.GetPage (printf "/%s/%s" (lower .field.key) $choice.value) }}{{ $href = .RelPermalink }}{{ end -}} {{- end -}} {{- with $choice -}} {{- $attrs := printf `data-choice="%s"` .value -}} {{- $hash := hash.FNV32a (printf "%s|%s" $.page.RelPermalink $.field.key) -}} {{- $style := printf "--bestiary-tilt: %.1fdeg; --bestiary-nudge: %dpx %.1fpx" (sub (div (mod $hash 450) 100.0) 3.0) (mul -1 (mod (div $hash 450) 7)) (sub (div (mod (div $hash 3150) 30) 10.0) 1.5) -}} {{- with .color }}{{ $style = printf "--bestiary-choice-color: %s; %s" . $style }}{{ end -}} {{- with or .nightColor .nightcolor }}{{ $style = printf "--bestiary-choice-night: %s; %s" . $style }}{{ $attrs = printf "%s data-night" $attrs }}{{ end -}} {{- $attrs = printf `%s style="%s"` $attrs ($style | safeCSS) -}} {{- if $href -}} {{ .label }} {{- else -}} {{ .label }} {{- end -}} {{- else -}} {{- errorf "%s: %s: %q is not one of the declared choices" .page.File.Path .field.key .value -}} {{- end -}}
