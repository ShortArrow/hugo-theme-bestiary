{{- /* Renders the declared choice whose value matches, linked to its page when the field has pages = true; fails the build when none does. */ -}} {{- $value := .value -}} {{- $choice := "" -}} {{- range .field.choices -}} {{- if eq .value $value }}{{ $choice = . }}{{ end -}} {{- end -}} {{- $href := "" -}} {{- if and $choice .field.pages -}} {{- with site.GetPage (printf "/%s/%s" (lower .field.key) $choice.value) }}{{ $href = .RelPermalink }}{{ end -}} {{- end -}} {{- with $choice -}} {{- $attrs := printf `data-choice="%s"` .value -}} {{- with .color }}{{ $attrs = printf `%s style="--bestiary-choice-color: %s"` $attrs (. | safeCSS) }}{{ end -}} {{- if $href -}} {{ .label }} {{- else -}} {{ .label }} {{- end -}} {{- else -}} {{- errorf "%s: %s: %q is not one of the declared choices" .page.File.Path .field.key .value -}} {{- end -}}
