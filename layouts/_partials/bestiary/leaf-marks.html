{{- /* Returns the custom properties that place a page's small handling marks: where its two thumb smudges and each foxing spot fall, and which spots the page lacks. The larger stains are drawn by bestiary/stains.html. Each value comes from its own hash of the page's path and the mark's name, so marks move independently of one another and each page keeps its own marks from build to build. */ -}} {{- $path := .RelPermalink -}} {{- $ranges := dict "smudge-left" (slice 10 90) "smudge-right" (slice 10 90) -}} {{- range seq 10 }} {{- $ranges = merge $ranges (dict (printf "fox-%d-x" .) (slice 3 97) (printf "fox-%d-y" .) (slice 3 97)) }} {{- end -}} {{- $props := slice -}} {{- range $name, $range := $ranges }} {{- $props = $props | append (printf "--mark-%s: %d%%" $name (partial "bestiary/roll.html" (dict "path" $path "name" $name "low" (index $range 0) "high" (index $range 1)))) }} {{- end -}} {{- range seq 10 }} {{- if eq (mod (hash.FNV32a (printf "%s#fox-%d?" $path .)) 3) 0 }} {{- $props = $props | append (printf "--mark-fox-%d: transparent" .) }} {{- end }} {{- end -}} {{- return delimit $props "; " -}}
