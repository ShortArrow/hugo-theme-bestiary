{{- /* Returns the custom properties that place the handling marks on a page's leaf: where its cup ring, tide mark, thumb smudges, blot and foxing fall, and whether it has a cup ring, a second ring and a tide mark at all. The values come from a hash of the page's path, so each page keeps its own marks from build to build. */ -}} {{- $a := hash.FNV32a .RelPermalink -}} {{- $b := hash.FNV32a (printf "%s#marks" .RelPermalink) -}} {{- $props := slice (printf "--mark-ring-x: %d%%" (add 55 (mod $a 38))) (printf "--mark-ring-y: %d%%" (add 5 (mod (div $a 38) 40))) (printf "--mark-tide-x: %d%%" (add 18 (mod (div $a 1520) 64))) (printf "--mark-tide-y: %d%%" (add 55 (mod (div $a 97280) 40))) (printf "--mark-smudge-left: %d%%" (add 15 (mod $b 60))) (printf "--mark-smudge-right: %d%%" (add 25 (mod (div $b 60) 60))) (printf "--mark-blot-x: %d%%" (add 6 (mod (div $b 3600) 40))) (printf "--mark-blot-y: %d%%" (add 20 (mod (div $b 144000) 70))) (printf "--mark-fox-x: %d%%" (sub (mod (div $b 10080000) 17) 8)) (printf "--mark-fox-y: %d%%" (sub (mod (div $a 3890000) 17) 8)) -}} {{- if eq (mod (div $a 66130000) 4) 0 }} {{- $props = $props | append "--mark-ring: transparent" "--mark-ring-2: transparent" }} {{- else if eq (mod (div $b 171000000) 2) 0 }} {{- $props = $props | append "--mark-ring-2: transparent" }} {{- end -}} {{- if eq (mod (div $a 264520000) 3) 0 }}{{ $props = $props | append "--mark-tide: transparent" }}{{ end -}} {{- return delimit $props "; " -}}
